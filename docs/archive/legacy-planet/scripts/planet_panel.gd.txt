func _fingerprint() -> String:
	var batches=[]
	for item in state.storage.batches: batches.append([item.id,item.quantity])
	var p=state.planet; var w=p.worlds.get_world()
	var research=[]
	for item in state.campus.research_jobs: research.append([item.id,int(item.progress)/5])
	return str([p.joined,p.qualified,p.extra_qualifications.keys(),p.reports.keys(),p.circuit_reports.keys(),p.materials.records.size(),p.materials.designs.size(),p.laminates.stock,p.laminates.feed,p.laminates.job.get("id",0),int(p.laminates.cooldown)/5,state.campus.unlocked_materials,research,p.feed,p.supplies,p.products.size(),p.job.get("id",0),w.id,w.paused,w.allowance,w.eco_allowance,w.zones,w.event_serial,int(w.seconds)/5,p.worlds.worlds.size(),batches])

func _redraw() -> void:
	if state==null: return
	UI.clear(body); UI.clear(world_tabs); laminate_progress=null; progress=null; progress_text=null; eco_widgets={}; watershed_widgets={}; village_widgets={}; habitat_widgets={}; industry_widgets={}; modern_widgets={}
	var p=state.planet
	for w in p.worlds.worlds:
		var button=UI.button(w.name,func(): _do("select",{"world_id":str(w.id)}),w.id==p.worlds.active_id)
		button.custom_minimum_size.x=166; world_tabs.add_child(button)
	if section=="dossiers": preload("res://scripts/player_material_panel.gd").build(self)
	elif section=="materials": preload("res://scripts/material_lab_panel.gd").build(self)
	elif section=="circuits": preload("res://scripts/planet_industry_panel.gd").circuit(self)
	elif section=="workshop": _workshop()
	elif view.display_mode=="watershed": preload("res://scripts/planet_watershed_panel.gd").build(self)
	elif view.display_mode=="village": preload("res://scripts/planet_village_panel.gd").build(self)
	elif view.display_mode=="modern": preload("res://scripts/planet_modern_panel.gd").build(self)
	elif view.display_mode=="industry": preload("res://scripts/planet_industry_panel.gd").build(self)
	elif view.display_mode=="habitat": preload("res://scripts/planet_habitat_panel.gd").build(self)
	elif view.display_mode=="ecology": EcologyPanel.build(self)
	else: _observations()
	fingerprint=_fingerprint()
	view.sync(display_snapshot())

func _scene(mode: String,id: String="") -> void:
	_cancel_gesture(); replay_index=-1; section="observe"
	if not id.is_empty(): selected_recipe=id
	view.set_mode(mode); view.reset_camera(); _redraw(); _live()

func _open_materials() -> void:
	material_return_mode=view.display_mode
	if material_return_mode=="modern": material_mode="same_mass"
	_cancel_gesture(); replay_index=-1; section="materials"
	if not state.planet.worlds.get_world().paused: state.planet.command(state,"pause")
	view.set_mode("materials"); view.reset_camera(); view.material_view.set_mode(material_mode)
	_redraw(); _live()

func _choose_recipe(id: String) -> void:
	_cancel_gesture(); selected_recipe=id; replay_index=-1
	if section in ["materials","circuits","dossiers"]: section="workshop"
	var domain=str(state.planet.recipe(id).get("domain",""))
	var use_watershed=domain=="watershed" or (domain.is_empty() and (view.display_mode=="watershed" or (view.display_mode=="materials" and material_return_mode=="watershed")))
	var use_village=domain=="village" or (id=="standard_water_crate" and view.display_mode=="village")
	var use_habitat=domain=="habitat" or (id=="standard_water_crate" and view.display_mode=="habitat")
	view.set_mode("modern" if domain=="modern" else "industry" if domain=="industry" else "habitat" if use_habitat else ("village" if use_village else ("watershed" if use_watershed else ("water" if id=="standard_water_crate" else "ecology"))))
	_redraw(); _live()

func display_snapshot() -> Dictionary:
	var world=state.planet.worlds.get_world()
	if replay_index<0: return world
	var copy=world.duplicate(); var history=_history()
	var entry=history[clampi(replay_index,0,history.size()-1)]
	copy.seconds=entry.at; copy.paused=true
	if view.display_mode=="watershed":
		copy.watershed=entry.state.duplicate(true); copy.watershed.history=history
	elif view.display_mode=="village":
		copy.village=entry.state.duplicate(true); copy.village.history=history
	elif view.display_mode=="modern":
		copy.modern=entry.state.duplicate(true); copy.modern.history=history
	elif view.display_mode=="industry":
		copy.industry=entry.state.duplicate(true); copy.industry.history=history
	elif view.display_mode=="habitat":
		copy.habitat=entry.state.duplicate(true); copy.habitat.history=history
	else:
		copy.ecology=world.ecology.duplicate(); copy.ecology.regions=entry.regions.duplicate(true)
	return copy

func _history() -> Array:
	var w=state.planet.worlds.get_world()
	if view.display_mode=="watershed": return w.watershed.history
	if view.display_mode=="village": return w.village.history
	if view.display_mode=="habitat": return w.habitat.history
	if view.display_mode=="industry": return w.industry.history
	if view.display_mode=="modern": return w.modern.history
	return w.ecology.history

func _replay() -> void:
	_cancel_gesture()
	if replay_index>=0: replay_index=-1
	else:
		var world=state.planet.worlds.get_world()
		if not world.paused: state.planet.command(state,"pause")
		replay_index=_history().size()-1
	_redraw(); _live()

func _begin_drag(at: Vector2) -> void:
	var p=state.planet; var w=p.worlds.get_world()
	if section in ["materials","circuits","dossiers"]: status.text="先在右侧记录对照，再选择制作构件"; return
	if replay_index>=0: status.text="回看中的世界不可投放，请先回到当前时刻"; return
	var count=p.available_in(w,selected_recipe)
	if not p.joined or count<1:
		status.text="先领取合作并完成装罐，或建立独立实验副本"; return
	dragging=true; armed=false; rotating=false; pointer=at; start=at
	token=p.shipment_serial; target_world=w.id; target_recipe=selected_recipe; ghost.show(); ghost.move_to_front()

func _tray_input(event: InputEvent) -> void:
	if event is InputEventScreenTouch and event.pressed:
		touch_index=event.index; _begin_drag(tray.global_position+event.position); accept_event()
	elif event is InputEventMouseButton and event.button_index==MOUSE_BUTTON_LEFT and event.pressed and touch_index<0:
		_begin_drag(get_global_mouse_position()); accept_event()

func _site_at(at: Vector2) -> String:
	if not viewport_box.get_global_rect().has_point(at): return ""
	return view.pick((at-viewport_box.global_position)*Vector2(viewport.size)/viewport_box.size)

func _end_drag(at: Vector2) -> void:
	var site=_site_at(at)
	if not site.is_empty():
		var data={"world_id":target_world,"site_id":site,"token":token,"recipe":target_recipe}
		_do("deploy",data)
	elif start.distance_to(at)<10 and tray.get_global_rect().has_point(at):
		dragging=false; armed=true; status.text="已选中成品，点击落点放置；返回键取消"
	else:
		_cancel_gesture(); status.text="已取消投放，成品没有消耗"

func _cancel_gesture() -> void:
	dragging=false; armed=false; rotating=false; touch_index=-1
	if is_instance_valid(ghost): ghost.hide()
	if is_instance_valid(view): view.highlighted=""

func _input(event: InputEvent) -> void:
	if state==null or closing: return
	var at=Vector2.ZERO; var press=false; var release=false; var motion=false
	if event is InputEventScreenTouch:
		at=event.position; press=event.pressed; release=not press
		if press and viewport_box.get_global_rect().has_point(at): touch_index=event.index
		elif event.index!=touch_index: return
	elif event is InputEventScreenDrag:
		if event.index!=touch_index: return
		at=event.position; motion=true
	elif event is InputEventMouseButton:
		if touch_index>=0: return
		at=event.position
		if viewport_box.get_global_rect().has_point(at) and event.pressed and event.button_index in [MOUSE_BUTTON_WHEEL_UP,MOUSE_BUTTON_WHEEL_DOWN]:
			view.zoom(-0.5 if event.button_index==MOUSE_BUTTON_WHEEL_UP else 0.5); get_viewport().set_input_as_handled(); return
		if event.button_index!=MOUSE_BUTTON_LEFT: return
		press=event.pressed; release=not press
	elif event is InputEventMouseMotion:
		if touch_index>=0: return
		at=event.position; motion=true
	else: return
	if press and viewport_box.get_global_rect().has_point(at):
		# Camera buttons are ordinary controls above the viewport.
		if at.y>=660 and at.x<610 and not armed: return
		if armed: dragging=true; armed=false
		else: rotating=true
		pointer=at; start=at; get_viewport().set_input_as_handled()
	elif motion and (dragging or armed or rotating):
		if rotating: view.rotate_view(at-pointer)
		pointer=at
		if dragging or armed: view.highlighted=_site_at(pointer)
		get_viewport().set_input_as_handled()
	elif release and dragging:
		_end_drag(at); touch_index=-1; get_viewport().set_input_as_handled()
	elif release and rotating:
		rotating=false; touch_index=-1; get_viewport().set_input_as_handled()

func _process(dt: float) -> void:
	if state==null or closing: return
	refresh+=dt
	if refresh>=0.25:
		refresh=0
		if fingerprint!=_fingerprint(): _redraw()
		_live()
	for id in site_labels:
		var badge: PanelContainer=site_labels[id]
		badge.visible=view.site_visible(id)
		badge.position=viewport_box.position+view.project_site(id)*viewport_box.size/Vector2(viewport.size)+Vector2(-badge.size.x*0.5,35 if id.begins_with("habitat_") or id.begins_with("industry_") or id.begins_with("modern_") else 85)
	if dragging or armed:
		ghost.position=pointer+Vector2(14,-40)
		view.highlighted=_site_at(pointer)
		view.highlight_valid=state.planet.worlds.can_deploy(target_world,view.highlighted,target_recipe)
		ghost.get_child(0).text="落点不适用" if not view.highlighted.is_empty() and not view.highlight_valid else "▣  "+str(state.planet.recipe(target_recipe).name)

func handle_back() -> bool:
	if dragging or armed: _cancel_gesture()
	else: _close()
	return true

func _close() -> void:
	if closing: return
	closing=true; _cancel_gesture(); state.planet.observing=false; dismissed.emit(); queue_free()

func _exit_tree() -> void:
	if state!=null and state.planet.observer_id==get_instance_id(): state.planet.observing=false

func _open_dossiers() -> void:
	_cancel_gesture(); replay_index=-1; section="dossiers"

	view.set_mode("dossiers"); view.reset_camera(); _redraw(); _live()
